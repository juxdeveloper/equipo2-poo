Clase Medico
    Atributos:
        nombre: Cadena
        cedula, especialidad, noPacientes: Entero
        atendiendo: Paciente (Nulo)
        sis: Sistema (Nulo)
        pacientesAsignados: Lista de Paciente

    Constructor(nombre, cedula, especialidad):
        Asignar atributos iniciales con noPacientes = 0

    Método registroEnSistema(sistema):
        sis = sistema
        sistema.registroMedico(esteMedico)

    Método solicitarPaciente(sistema) -> Booleano:
        Si noPacientes < 10 Entonces:
            p = sistema.asignarPaciente(esteMedico, especialidad)
            Si p != Nulo Entonces:
                Agregar p a pacientesAsignados
                p.setMedico(esteMedico)
                noPacientes = noPacientes + 1
                Retornar Verdadero
            FinSi
        FinSi
        Retornar Falso

    Método darConsulta(paciente) -> Booleano:
        Si paciente está en pacientesAsignados Y atendiendo == Nulo Entonces:
            atendiendo = paciente
            paciente.setEstado(EN_CONSULTA)
            Retornar Verdadero
        FinSi
        Retornar Falso

    Método darTratamiento(paciente, prescripcion) -> Booleano:
        Si paciente == atendiendo Y sis.asignarPaciente(paciente, esteMedico, especialidad, prescripcion) Entonces:
            atendiendo = Nulo
            paciente.setTratamiento(prescripcion)
            paciente.setEstado(EN_HOSPITAL)
            Retornar Verdadero
        FinSi
        Retornar Falso

    Método verListaPacientes():
        Ordenar pacientesAsignados por apellido de forma A-Z
        Para cada p En pacientesAsignados Hacer:
            Imprimir p.apellido + " " + p.nombre + " - Especialidad: " + p.especialidad
FinClase
